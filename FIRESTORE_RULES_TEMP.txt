// TEMPORARY RULES FOR SEEDING - USE ONLY WHILE RUNNING SEED SCRIPT
// After seeding is complete, revert to production rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId} {
      allow read, write: if request.auth != null && (
        request.auth.uid == userId ||
        request.auth.token.email == userId
      );

      match /shooting-sessions/{sessionId} {
        allow read, write: if request.auth != null && (
          request.auth.uid == userId ||
          request.auth.token.email == userId
        );
      }

      match /drills/{drillId} {
        allow read, write: if request.auth != null && (
          request.auth.uid == userId ||
          request.auth.token.email == userId
        );
      }

      match /challengeProgress/{challengeId} {
        allow read, write: if request.auth != null && (
          request.auth.uid == userId ||
          request.auth.token.email == userId
        );

        match /drillAttempts/{drillId} {
          allow read, write: if request.auth != null && (
            request.auth.uid == userId ||
            request.auth.token.email == userId
          );
        }
      }

      match /badges/{badgeId} {
        allow read, write: if request.auth != null && (
          request.auth.uid == userId ||
          request.auth.token.email == userId
        );
      }
    }

    match /user-scores/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && (
        request.auth.uid == userId ||
        request.auth.token.email == userId
      );
    }

    // TEMPORARY: Allow writes for seeding
    match /challenges/{challengeId} {
      allow read: if true;
      allow write: if true; // ⚠️ TEMPORARY - Change to "false" after seeding!

      match /drills/{drillId} {
        allow read: if true;
        allow write: if true; // ⚠️ TEMPORARY - Change to "false" after seeding!
      }
    }

    match /series/{seriesId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    match /leaderboards/{leaderboardId} {
      allow read: if true;
      allow write: if false;
    }

    // Drill Sessions - users can read their own sessions
    match /drillSessions/{sessionId} {
      allow read: if request.auth != null && request.auth.uid == resource.data.uid;
      allow create: if request.auth != null && request.auth.uid == request.resource.data.uid;
      allow update, delete: if request.auth != null && request.auth.uid == resource.data.uid;
    }
  }
}
